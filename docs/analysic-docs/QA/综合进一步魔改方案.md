  新增函数（在文件末尾添加）：

  def get_token_type(token: str | None) -> str:
      """
      检测 Token 类型。
      
      Returns:
          "oauth" - OAuth Token (sk-ant-oat01-)
          "api_key" - API Key (sk-ant-api-)
          "enterprise" - Enterprise/CCR Token
          "unknown" - 未知类型
      """
      if not token:
          return "unknown"

      if token.startswith("sk-ant-oat01-") or token.startswith("sk-ant-oat"):
          return "oauth"
      elif token.startswith("sk-ant-api") or token.startswith("sk-"):
          return "api_key"
      elif token.startswith("ccr-") or "anthropic" in token.lower():
          return "enterprise"
      else:
          return "unknown"


  def is_api_key_mode() -> bool:
      """检查当前是否使用 API Key 模式。"""
      token = get_auth_token()
      return get_token_type(token) == "api_key"
